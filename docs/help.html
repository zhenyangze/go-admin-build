框架概述

GoAdmin 是一个受 Dcat Admin 启发的 Go 语言后台管理框架。它提供了资源导向的 DSL（领域特定语言），用于快速构建管理后台界面。

核心特性

  • 资源导向设计：以资源为中心，自动构建 CRUD 界面
  • 纯 HTTP 标准：基于 net/http，可嵌入任何 Web 框架（Gin、Echo、Chi 等）
  • 丰富的字段类型：支持文本、选择、文件上传、富文本、关联等多种字段
  • 内置认证授权：完整的 RBAC 权限管理系统
  • 响应式 UI：基于 TailwindCSS 的现代化界面
  • GORM 集成：内置 GORM 数据仓库，支持多种数据库

安装与配置

安装依赖

go get github.com/zhenyangze/goadmin

npm run build:css

CSS 构建是必需的，因为框架使用 TailwindCSS 生成样式。每次修改 input.css 后都需要重新运行 npm run build:css。

快速开始

1. 定义模型

package models

import "gorm.io/gorm"

type Article struct {
    gorm.Model
    Title   string `gorm:"size:200;not null"`
    Content string `gorm:"type:text"`
    Status  int    `gorm:"default:1"`
}

2. 创建资源

package resources

import (
    "github.com/zhenyangze/goadmin"
    "github.com/zhenyangze/goadmin/form"
    "github.com/zhenyangze/goadmin/grid"
    "myapp/models"
)

func ArticleResource() goadmin.Resource {
    repo := gormstore.NewRepository[models.Article](db, "articles")

    return goadmin.Resource{
        Name:       "文章管理",
        Path:       "articles",
        Repository: repo,
        Grid: grid.New(func(g *grid.Builder) {
            g.Column("id", "ID")
            g.Column("title", "标题")
            g.Column("status", "状态")
            g.RowActions(
                grid.ActionEdit(),
                grid.ActionDelete(),
            )
        }),
        Form: form.New(func(f *form.Builder) {
            f.Text("title", "标题").Required()
            f.Textarea("content", "内容")
            f.Select("status", "状态").
                Options(map[string]string{
                    "1": "启用",
                    "0": "禁用",
                })
        }),
    }
}

3. 启动应用

package main

import (
    "log"
    "net/http"

    "github.com/zhenyangze/goadmin"
    "myapp/resources"
)

func main() {
    app := goadmin.New(goadmin.Config{
        Title:         "我的博客后台",
        SessionSecret: "your-secret-key",
    })

    // 注册资源
    app.Register(resources.ArticleResource())

    // 启动服务
    log.Println("服务启动: http://localhost:8080/admin")
    log.Fatal(http.ListenAndServe(":8080", app.Handler()))
}

核心概念

Resource（资源）

Resource 是 GoAdmin 的核心抽象，代表一个可管理的业务实体。每个资源对应数据库中的一张表，框架会自动为其生成列表、创建、编辑、删除等页面。

Builder（构建器）

GoAdmin 使用构建器模式来配置页面：

  • Grid.Builder - 配置列表页
  • Form.Builder - 配置表单页
  • Show.Builder - 配置详情页
  • Tree.Builder - 配置树形页

Repository（数据仓库）

Repository 接口抽象了数据访问层，框架内置了 GORM 实现，你也可以自定义实现。

资源定义

Resource 结构

字段 类型 说明
Name string 资源显示名称，用于菜单和页面标题
Path string URL 路径，如 "articles"
Repository Repository 数据仓库接口实现
Grid GridBuilder 列表页配置（可选）
Form FormBuilder 表单页配置（可选）
Show ShowBuilder 详情页配置（可选）
Tree TreeBuilder 树形页配置（可选）

完整示例

goadmin.Resource{
    Name:       "文章管理",
    Path:       "articles",
    Icon:       "file-text",  // 菜单图标

    // 数据层
    Repository: articleRepo,

    // 列表页
    Grid: grid.New(func(g *grid.Builder) {
        g.Column("id", "ID")
        g.Column("title", "标题").Searchable()
        g.Column("category.name", "分类")
        g.Column("created_at", "创建时间").Sortable()

        g.Filter("status", "状态").Select(...)
        g.SearchableFields("title", "content")

        g.PageActions(grid.ActionCreate())
        g.RowActions(
            grid.ActionView(),
            grid.ActionEdit(),
            grid.ActionDelete(),
        )
    }),

    // 表单页
    Form: form.New(func(f *form.Builder) {
        f.Text("title", "标题").Required().Rules("min:3,max:100")
        f.Select("category_id", "分类").FromResource("categories")
        f.Editor("content", "内容")
        f.Switch("status", "状态").Default(true)
    }),

    // 详情页
    Show: show.New(func(s *show.Builder) {
        s.Field("id", "ID")
        s.Field("title", "标题")
        s.Field("content", "内容").AsHtml()
        s.Field("created_at", "创建时间")
    }),
}

Grid 列表页

Grid 用于配置数据列表展示，支持列定义、筛选、搜索、排序、分页等功能。

基础列定义

g.Grid: grid.New(func(g *grid.Builder) {
    // 简单列
    g.Column("id", "ID")
    g.Column("title", "文章标题")

    // 可排序列
    g.Column("created_at", "创建时间").Sortable()

    // 可搜索列
    g.Column("title", "标题").Searchable()

    // 关联列（自动解析嵌套字段）
    g.Column("category.name", "分类名称")
    g.Column("author.email", "作者邮箱")
}),

列方法

Column(name, label string) *ColumnBuilder
定义数据列
name: 字段名（支持点号嵌套，如 "user.name"）
label: 显示标题
Sortable()
启用列排序功能
Searchable()
启用列搜索功能

筛选器

g.Filter("status", "状态").Select(map[string]string{
    "1": "启用",
    "0": "禁用",
})

g.Filter("created_at", "创建时间").DateRange()

操作按钮

// 页面级操作（顶部）
g.PageActions(
    grid.ActionCreate().Label("新建文章"),
    grid.ActionExport(),
)

// 行级操作
// 默认提供：查看、编辑、删除
g.RowActions(
    grid.ActionView().Label("查看详情"),
    grid.ActionEdit(),
    grid.ActionDelete().Confirm("确定删除吗？"),

    // 自定义操作
    grid.Action{
        Label: "审核",
        Style: "success",
        URL:   "/admin/articles/{id}/approve",
    },
)

分页配置

g.Paginate(20)  // 每页 20 条（默认 15）

Form 表单页

Form 用于配置创建和编辑页面，提供丰富的表单字段类型和验证规则。

基础字段

文本字段

f.Text("name", "名称").
    Required().
    Rules("min:2,max:50").
    Placeholder("请输入名称").
    Help("2-50 个字符")

f.Textarea("description", "描述").Rows(4)

f.Password("password", "密码")

选择字段

// 下拉选择
f.Select("status", "状态").
    Options(map[string]string{
        "active":   "启用",
        "inactive": "禁用",
        "pending":  "待审核",
    }).
    Default("active")

// 多选
f.MultiSelect("tags", "标签").
    Options(tagOptions)

// 开关
f.Switch("is_featured", "推荐").Default(false)

日期时间

// 日期
f.Date("start_date", "开始日期")

// 日期时间
f.DateTime("publish_at", "发布时间")

// 日期范围（两个字段）
f.DateRange("start", "end", "有效期")

文件上传

// 单图上传
f.Image("cover", "封面图").
    Dir("uploads/images").
    MaxSize(2 * 1024 * 1024).  // 2MB
    AllowedExt("jpg", "png", "gif")

// 多图上传
f.Gallery("images", "图集").
    Dir("uploads/gallery")

// 文件
f.File("attachment", "附件")

高级表单字段

关联字段

//  belongsTo 关联
f.Select("category_id", "分类").
    FromResource("categories")  // 从分类资源加载选项

// 多对多关联
f.MultiSelect("tags", "标签").
    FromResource("tags")

富文本编辑器

f.Editor("content", "内容").
    Height(400)

标签输入

f.Tags("keywords", "关键词")

// 带建议
tagField := f.Tags("labels", "标签")
tagField.Suggestions = []string{"重要", "紧急", "日常"}

Repeater（重复子表单）

用于管理一对多关联数据，如文章的 FAQ、产品的规格参数等。

f.Repeater("faqs", "常见问题").
    Fields(
        form.Text("question", "问题").Required(),
        form.Textarea("answer", "回答").Required(),
    )

// 使用自定义仓库处理
repo := gormstore.NewRepeaterRepository(db, "articles", "faqs")

分组显示

f.Tab("基本信息", func(t *form.TabBuilder) {
    t.Text("title", "标题")
    t.Select("category", "分类")
})

f.Tab("高级设置", func(t *form.TabBuilder) {
    t.Switch("is_vip", "VIP 专属")
    t.Number("sort", "排序权重")
})

验证规则

规则 说明 示例
required 必填 Required()
min/max 长度限制 Rules("min:3,max:100")
email 邮箱格式 Rules("email")
numeric 纯数字 Rules("numeric")
url URL 格式 Rules("url")

Show 详情页

Show 用于展示单条记录的详细信息。

基础用法

s.Show: show.New(func(s *show.Builder) {
    s.Field("id", "ID")
    s.Field("title", "标题")
    s.Field("content", "内容").AsHtml()  // HTML 渲染
    s.Field("created_at", "创建时间")
}),

字段方法

AsHtml()
字段值作为 HTML 渲染（不进行转义）
AsImage()
将值显示为图片
AsLink()
显示为可点击链接
Badge(style string)
显示为徽章标签
style: "success" | "warning" | "danger" | "info"

Tree 树形页

Tree 用于展示和编辑层级结构数据，如分类、菜单、部门等。

数据模型

type Category struct {
    gorm.Model
    ParentID *uint      // 父节点 ID，根节点为 null
    Name     string
    Sort     int        // 排序权重
}

Tree 配置

s.Tree: tree.New(func(t *tree.Builder) {
    t.TitleField("name")        // 显示字段
    t.SortField("sort")         // 排序字段

    // 节点操作
    t.CanCreate()               // 允许创建子节点
    t.CanEdit()                 // 允许编辑
    t.CanDelete()               // 允许删除
    t.CanReorder()              // 允许拖拽排序
}),

自定义节点显示

t.Node(func(n *tree.Node, item *Category) {
    n.Title = item.Name
    n.Description = fmt.Sprintf("文章数: %d", item.ArticleCount)
    n.Icon = "folder"
})

认证授权

GoAdmin 内置完整的 RBAC（基于角色的访问控制）系统。

默认用户模型

// 框架自动提供以下表
// - admin_users      管理员用户
// - admin_roles      角色
// - admin_permissions 权限
// - admin_role_permissions 角色权限关联
// - admin_user_roles  用户角色关联

配置认证

authService := auth.NewService(db)

app := goadmin.New(goadmin.Config{
    AuthService: authService,
})

// 自定义导航（可选）
authService.SetNavigationFunc(func(ctx context.Context, identity *goadmin.Identity) []goadmin.NavigationItem {
    return []goadmin.NavigationItem{
        {Title: "仪表盘", Icon: "home", URI: "/admin"},
        {Title: "文章管理", Icon: "file-text", URI: "/admin/articles"},
        {Title: "系统设置", Icon: "settings", Children: []goadmin.NavigationItem{
            {Title: "用户管理", URI: "/admin/users"},
            {Title: "角色权限", URI: "/admin/roles"},
        }},
    }
})

资源权限控制

// 在资源中设置权限
res := goadmin.Resource{
    Name: "文章管理",
    Path: "articles",

    // 查看权限标识
    ViewPermission: "article.read",

    // 创建权限标识
    CreatePermission: "article.create",

    // 编辑权限标识
    EditPermission: "article.update",

    // 删除权限标识
    DeletePermission: "article.delete",
}

菜单权限

// 只为特定角色显示菜单项
navItem := goadmin.NavigationItem{

生命周期钩子

在数据操作的各个阶段执行自定义逻辑。

Repository 钩子

repo := gormstore.NewRepository[Article](db, "articles")

// 创建前
repo.BeforeCreate(func(ctx context.Context, values goadmin.Values) error {
    // 自动设置作者
    values["author_id"] = []string{fmt.Sprint(GetCurrentUserID(ctx))}
    return nil
})

// 创建后
repo.AfterCreate(func(ctx context.Context, id string, model *Article) error {
    // 发送通知
    notification.Send("新文章发布: " + model.Title)
    return nil
})

// 更新前
repo.BeforeUpdate(func(ctx context.Context, id string, values goadmin.Values) error {
    // 记录修改日志
    audit.Log("article", id, "update", values)
    return nil
})

// 删除前（软删除检查）
repo.BeforeDelete(func(ctx context.Context, id string) error {
    // 检查是否有关联数据
    if hasRelatedData(id) {
        return errors.New("该文章有关联数据，无法删除")
    }
    return nil
})

部署指南

生产环境配置

app := goadmin.New(goadmin.Config{
    Title:         "管理后台",
    SessionSecret: os.Getenv("SESSION_SECRET"),  // 从环境变量读取

    // 生产环境设置
    Debug: false,

    // 上传配置
    UploadDir: "/var/www/uploads",

    // 自定义错误页面
    ErrorHandler: func(err error, w http.ResponseWriter, r *http.Request) {
        // 记录错误日志
        log.Printf("Error: %v", err)
        http.Error(w, "Internal Server Error", 500)
    },
})

Docker 部署

# Dockerfile
FROM golang:1.21-alpine AS builder
WORKDIR /app
COPY . .
RUN go build -o admin main.go

FROM alpine:latest
RUN apk --no-cache add ca-certificates
WORKDIR /root/
COPY --from=builder /app/admin .
COPY --from=builder /app/goadmin/assets ./assets
EXPOSE 8080
CMD ["./admin"]
安全提醒
  • 生产环境务必使用强 SESSION_SECRET
  • 启用 HTTPS，配置安全的 Cookie
  • 限制上传文件大小和类型
  • 定期更新依赖包
